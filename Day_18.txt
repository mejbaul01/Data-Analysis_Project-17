CREATE TABLE Sales(
OrderId INT PRIMARY KEY,
Product VARCHAR(50),
Region	VARCHAR(50),
Quantity INT, 
Revenue DECIMAL (10,2)

);

SELECT * FROM Sales;


SELECT Product,
Count(*) AS Total_Order,
SUM(Revenue) AS Total_Revenue,
AVG(Revenue) AS Average_Revenue
FROM Sales
GROUP BY Product;
